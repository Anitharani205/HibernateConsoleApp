CREATE TABLE BORROWER_TBL (
    borrowerID NUMBER PRIMARY KEY,
    fullName VARCHAR2(100),
    gender VARCHAR2(10),
    dateOfBirth DATE,
    villageOrArea VARCHAR2(100),
    riskCategory VARCHAR2(20),
    status VARCHAR2(20)
);
CREATE SEQUENCE BORROWER_SEQ START WITH 1 INCREMENT BY 1 NOCACHE NOCYCLE;

CREATE TABLE LOAN_TBL (
    loanID NUMBER PRIMARY KEY,
    borrowerID NUMBER REFERENCES BORROWER_TBL(borrowerID),
    principalAmount FLOAT(10),
    interestRate FLOAT(5),
    status VARCHAR2(20)
);
CREATE SEQUENCE LOAN_SEQ START WITH 1 INCREMENT BY 1 NOCACHE NOCYCLE;

CREATE TABLE INSTALLMENT_TBL (
    installmentID NUMBER PRIMARY KEY,
    loanID NUMBER REFERENCES LOAN_TBL(loanID),
    dueAmount FLOAT(10),
    status VARCHAR2(20)
);
CREATE SEQUENCE INSTALLMENT_SEQ START WITH 1 INCREMENT BY 1 NOCACHE NOCYCLE;
